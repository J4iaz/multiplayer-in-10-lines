Multiplayer in 10 Lines — Live Proof

The counter on the right is connected live using the exact 10 lines of code below. Click the button or open in another tab to see it sync in real time.

multiplayer-demo.js 
// 1. Drop in script tag: <script src="/multiplayer.js"></script>
const room = Multiplayer.join({ room: 'demo' });

// 2. Listen for connection status updates
room.onStatusChange(s => setStatus(s.status));

// 3. Reactive state update listener across all browsers
room.onStateChange(state => {
  btn.textContent = `Clicked ${state.clicks || 0} times`;
});

// 4. Update shared state — broadcasts instantly
btn.onclick = () => {
  room.setState('clicks', (room.getState('clicks') || 0) + 1);
};

// 5. Enable collaborative live cursors
room.enableCursors();
